} } // ═══════════════ 時段工具 ═══════════════ /** 'HH:MM' → 分鐘數 */ function hs_toMin_(t) { const m = String(t || '').match(/(\d{1,2}):(\d{2})/); if (!m) return -1; return Number(m[1]) * 60 + Number(m[2]); } } return { ok: true }; } catch (err) { writeLog('hs_checkSlots_', err.message); return { ok: true }; // 查不到就放行，寧可讓 Elsa 手動處理也不要擋掉客人 } } /** * v1.1 客人想約公休／休息日時通知 Elsa（排班需求訊號） * 同一個日期 6 小時內只推一次，避免同一天被反覆嘗試而洗頻。 * 撞單與手動鎖定不推播（每天都會發生，對排班沒有意義）。 */ function hs_notifyBlockedClosed_(info) { try { const cache = CacheService.getScriptCache(); pushLine(OWNER_UID, msg); } catch (err) { writeLog('hs_notifyBlockedClosed_', err.message); } } // ═══════════════ 主收單 ═══════════════ /** * action=homeBooking * 參數：name, phone, address, date, time, adults, kids, services, parking, uid, displayName, note */ function handleHomeBooking(e) { const lock = LockService.getScriptLock(); try { if (!hs_isEnabled_()) { return okJson({ status: 'error', msg: 'disabled', text: '線上到府預約暫時關閉，請透過 LINE 與 Elsa 聯繫 🙏' }); } const p = (e && e.parameter) ? e.parameter : {}; const name = String(p.name || '').trim(); const kids = String(p.kids || '0').trim(); const svc = String(p.services || '').trim(); const parking = String(p.parking || '').trim(); const note = String(p.note || '').trim(); const uid = String(p.uid || '').trim(); const dName = String(p.displayName || '').trim(); // 必填檢查 if (!name || !phone || !address || !dateStr || !timeStr || !svc) { return okJson({ status: 'error', msg: 'missing_fields', text: '資料未填完整，請確認姓名／電話／地址／日期／時段／服務項目' }); } // 電話格式 const digits = phone.replace(/\D/g, ''); if (digits.length < 8) { return okJson({ status: 'error', msg: 'bad_phone', text: '請填寫正確的聯絡電話' }); } lock.waitLock(20000); const count = hs_slotCount_();
